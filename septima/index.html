Personnes gagnant plus de 2000$

list_gagne_bien = []
for person in people:
    if float(person["income"].replace("$", "")) > 2000:
        list_gagne_bien.append(person)

print(len(list_gagne_bien))

Documentaires + salaire > 1482$

nb_pers_aiment_doc_plus = []
salaire_min = 1482

for person in people:
    if float(person["income"].replace("$", "")) > salaire_min:
        if "Documentary" in person["pref_movie"]:
            nb_pers_aiment_doc_plus.append(person)

print(len(nb_pers_aiment_doc_plus))

Personnes gagnant plus de 4000$ (infos réduites)

list_peoples = []
min_salary = 4000

for person in people:
    if float(person["income"].replace("$", "")) > min_salary:
        list_peoples.append({
            "name": person["last_name"],
            "first_name": person["first_name"],
            "id": person["id"],
            "income": person["income"]
        })

print(list_peoples)

Homme le plus riche

person_most_rich = {}
more_rich = 0

for person in people:
    value = float(person["income"].replace("$", ""))
    if more_rich < value:
        more_rich = value
        person_most_rich = {
            "nom": person["last_name"],
            "id": person["id"]
        }

print(person_most_rich)

Salaire moyen

all_salary = []

for person in people:
    value = float(person["income"].replace("$", ""))
    all_salary.append(value)

salaire_moyen = np.array(all_salary).mean()
print(salaire_moyen)
